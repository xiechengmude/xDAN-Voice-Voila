# Voila 全双工语音 API 服务测试计划

## 1. 测试目标

本测试计划旨在验证 Voila 全双工语音 API 服务的功能完整性、性能表现和稳定性。测试将覆盖以下方面：

- API 端点功能验证
- WebSocket 全双工通信
- 模型推理正确性
- 会话管理机制
- 错误处理和恢复
- 性能和负载测试

## 2. 测试环境

### 2.1 硬件环境
- 服务器: 配备 NVIDIA GPU (至少 8GB 显存)
- 客户端: 标准开发机器

### 2.2 软件环境
- 操作系统: Ubuntu 20.04 LTS / macOS
- Python 版本: 3.11.11
- 依赖库: 按照 requirements-api.txt 安装
- 测试工具: pytest, locust, curl, WebSocket 客户端

```python
# tests/utils/test_audio.py
import pytest
import numpy as np
from app.utils.audio import resample_audio, encode_audio_base64, decode_audio_base64

def test_resample_audio():
    # 创建测试音频数据
    sample_rate = 44100
    target_rate = 16000
    duration = 1  # 1秒
    t = np.linspace(0, duration, sample_rate * duration)
    audio_data = np.sin(2 * np.pi * 440 * t)  # 440Hz 正弦波
    
    # 测试重采样
    resampled = resample_audio(audio_data, sample_rate, target_rate)
    
    # 验证结果
    assert len(resampled) == target_rate * duration
    
def test_audio_base64_conversion():
    # 创建测试音频数据
    sample_rate = 16000
    duration = 1  # 1秒
    t = np.linspace(0, duration, sample_rate * duration)
    audio_data = np.sin(2 * np.pi * 440 * t)  # 440Hz 正弦波
    
    # 测试编码和解码
    audio_base64 = encode_audio_base64(audio_data, sample_rate)
    decoded_audio, decoded_rate = decode_audio_base64(audio_base64)
    
    # 验证结果
    assert decoded_rate == sample_rate
    np.testing.assert_allclose(audio_data, decoded_audio, rtol=1e-4)
```

```python
import pytest
from fastapi.testclient import TestClient
import json
import numpy as np
from unittest.mock import patch

from app.main import app

@pytest.fixture
def websocket_client():
    with TestClient(app) as client:
        yield client

def test_websocket_connection(websocket_client):
    # 首先创建会话
    response = websocket_client.post(
        "/api/session/create",
        json={"instruction": "测试指令", "voice_name": "Homer Simpson"}
    )
    session_id = response.json()["session_id"]
    
    # 模拟处理音频
    with patch("app.services.model_service.process_audio") as mock_process:
        mock_process.return_value = {
            "text": "模拟响应文本",
            "audio_base64": "base64_encoded_audio"
        }
        
        # 连接 WebSocket
        with websocket_client.websocket_connect(f"/ws/{session_id}") as websocket:
            # 接收欢迎消息
            welcome = websocket.receive_json()
            assert welcome["type"] == "welcome"
            
            # 发送音频数据
            audio_data = np.zeros(1000).tolist()
            websocket.send_json({
                "type": "audio",
                "data": audio_data
            })
            
            # 接收响应
            response = websocket.receive_json()
            assert response["type"] == "response"
            assert response["data"]["text"] == "模拟响应文本"
            
            # 发送声音更改请求
            websocket.send_json({
                "type": "change_voice",
                "voice_name": "Marge Simpson"
            })
            
            # 接收信息响应
            info = websocket.receive_json()
            assert info["type"] == "info"
            assert "已切换到" in info["message"]
```

## 6. 性能测试

使用 Locust 进行负载测试：

```python
# tests/performance/locustfile.py
from locust import HttpUser, task, between
import json
import base64
import numpy as np
import io
import soundfile as sf

class VoilaApiUser(HttpUser):
    wait_time = between(1, 3)
    session_id = None
    
    def on_start(self):
        # 创建会话
        response = self.client.post(
            "/api/session/create",
            json={"instruction": "测试指令", "voice_name": "Homer Simpson"}
        )
        self.session_id = response.json()["session_id"]
    
    def on_stop(self):
        # 关闭会话
        if self.session_id:
            self.client.post(
                "/api/session/close",
                data={"session_id": self.session_id}
            )
    
    @task(3)
    def process_text(self):
        if not self.session_id:
            return
            
        self.client.post(
            "/api/text/process",
            json={
                "session_id": self.session_id,
                "text": "这是一个测试文本"
            }
        )
    
    @task(1)
    def process_audio(self):
        if not self.session_id:
            return
            
        # 创建测试音频
        sample_rate = 16000
        duration = 1  # 1秒
        t = np.linspace(0, duration, sample_rate * duration)
        audio_data = np.sin(2 * np.pi * 440 * t)  # 440Hz 正弦波
        
        # 编码为 base64
        audio_bytes = io.BytesIO()
        sf.write(audio_bytes, audio_data, sample_rate, format='WAV')
        audio_bytes.seek(0)
        audio_base64 = base64.b64encode(audio_bytes.read()).decode('utf-8')
        
        self.client.post(
            "/api/audio/process",
            json={
                "session_id": self.session_id,
                "audio_base64": audio_base64
            }
        )
```
